       ```typescript
       import { extractText, getDocumentProxy } from 'unpdf'

       export const MAX_FILE_BYTES = 10 * 1024 * 1024 // 10 MB — RESEARCH §A3

       export type PdfExtractResult =
         | { ok: true; text: string }
         | { ok: false; error: string }

       export async function extractPdfText(filePath: string): Promise<PdfExtractResult> {
         // 1. Size cap — stat before reading (RESEARCH Pitfall §5)
         const stat = await fs.stat(filePath)
         if (stat.size > MAX_FILE_BYTES) {
           return {
             ok: false,
             error: `File too large (${(stat.size / 1024 / 1024).toFixed(1)} MB; max 10 MB)`,
           }
         }
         // 2. Read as Buffer (NO encoding — RESEARCH Pitfall §4)
         const buffer = await fs.readFile(filePath)
         try {
           const pdf = await getDocumentProxy(new Uint8Array(buffer))
           const { text } = await extractText(pdf, { mergePages: true })
           return { ok: true, text: typeof text === 'string' ? text : text.join('\n') }
         } catch (err: any) {
           return { ok: false, error: err?.message ?? 'Could not extract text from PDF' }
         }
       }
       ```

       ```typescript
         const result = await dialog.showOpenDialog(win, {

           properties: ['openFile'],
           filters: [{ name: 'CV', extensions: ['md', 'pdf'] }],
         })
         if (result.canceled || result.filePaths.length === 0) {

         }
         const filePath = result.filePaths[0]
         const ext = path.extname(filePath).toLowerCase()

           if (ext === '.md') {
             const stat = await fs.stat(filePath)
             if (stat.size > MAX_FILE_BYTES) {
               return {
                 cancelled: false,
                 error: `File too large (${(stat.size / 1024 / 1024).toFixed(1)} MB; max 10 MB)`,

             return { cancelled: false, type: 'md' as const, content }
           }
           if (ext === '.pdf') {
             const result = await extractPdfText(filePath)
             if (!result.ok) {
               return { cancelled: false, error: result.error }
             }
             return { cancelled: false, type: 'pdf' as const, content: result.text }
           }
           return { cancelled: false, error: 'Unsupported file type. Choose a .md or .pdf file.' }
         } catch (err: any) {
           return { cancelled: false, error: err?.message ?? 'Could not read file. Check it exists and try again.' }
         }
       })

         try {
           const content = UpdateCvSchema.parse(raw)
           const cvPath = path.join(projectRoot, 'cv.md')
           // Pitfall §3: seed empty cv.md if missing, then lockAndWrite

           await lockAndWrite(cvPath, () => content, pendingGuiWrites)
           return { success: true }

           return { success: false, error: err?.message ?? 'Failed to write cv.md' }

         }
       })
       ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| renderer→preload→main (IPC) | All file picker, file reads, PDF extraction, and writes cross from sandboxed renderer into the main process via contextBridge. The renderer never sees a file path or buffer; it sends a content string for `updateCv` and receives content/cancellation/error for `openCvFilePicker`. |
| OS file dialog→main | The user picks an arbitrary file via the native OS dialog. The path returned is whatever the user chose. |
| External file content (PDF bytes) → unpdf parser | Untrusted bytes (anything the user picks) flow through pdfjs (bundled inside unpdf) for parsing. |
| main→cv.md (filesystem write) | A renderer-supplied content string crosses into a write to the project-canonical CV file. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08-01 | Tampering / Elevation of Privilege | `updateCv` write target | mitigate | Path is hard-coded to `path.join(projectRoot, 'cv.md')` inside the handler — renderer cannot redirect the write. Verified pattern (RESEARCH §"Known Threat Patterns"). |
| T-08-02 | Tampering | `openCvFilePicker` returned filePath | accept | Renderer never sees the path. Path stays inside main; the user chose it via OS-trusted dialog. No traversal vector across IPC. |
| T-08-03 | Denial of Service | `extractPdfText` (memory blowup on huge PDF) | mitigate | `fs.stat` before `fs.readFile`; reject files over `MAX_FILE_BYTES = 10 MB` (RESEARCH §A3 + Pitfall §5). Returns discriminated error result, never throws. |
| T-08-04 | Denial of Service | `updateCv` (huge markdown payload from a hostile renderer) | mitigate | `UpdateCvSchema = z.string().min(1).max(2_000_000)` enforces a 2 MB cap before lockAndWrite is called (RESEARCH §A4). |
| T-08-05 | Tampering / Information Disclosure | Malicious PDF triggering pdfjs CVE | mitigate | Out-of-renderer extraction (`pdf-extract` runs in main process — sandboxed renderer is isolated). unpdf bundles pdfjs and is current as of April 2026. Future CVEs handled by bumping unpdf via Dependabot. |
| T-08-06 | Tampering | File-extension spoofing (`.pdf` containing non-PDF bytes) | accept | We never EXECUTE uploaded content; we extract text. unpdf fails gracefully on invalid bytes; the discriminated result returns `{ ok: false, error: <message> }` — surfaced to the user as the standard "Could not extract text" error. |
| T-08-07 | Spoofing | Phishing dialog (UI redress) on `dialog.showOpenDialog` | accept | Native OS dialog cannot be redressed by web content (RESEARCH §"Known Threat Patterns"). |
| T-08-08 | Repudiation | Concurrent write to cv.md | mitigate | `lockAndWrite` uses `proper-lockfile` with stale=10s + retries (write-queue.ts existing pattern); concurrent writers serialize. |
| T-08-09 | Information Disclosure | Error messages leaking absolute filesystem paths into renderer | accept | Error strings from `unpdf` and `fs` may include paths; the user picked the file so they already know its path. No cross-user leak in single-user desktop app context. |
| T-08-10 | Denial of Service | TOCTOU between `fs.stat` and `fs.readFile` (file replaced after size check) | accept | Single-user desktop app; the window between stat and read is microseconds. Negligible threat per RESEARCH §"Known Threat Patterns". |
