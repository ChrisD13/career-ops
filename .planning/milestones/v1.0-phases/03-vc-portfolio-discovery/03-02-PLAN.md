  ```typescript
  import cron from 'node-cron'
  // ...existing imports

  const DEFAULT_MODEL = 'claude-sonnet-4-6'
  export const DEFAULT_VC_INTERVAL = '0 9 1 * *'   // monthly, 1st of month at 09:00 local
  const ALLOWED_MODELS = ['claude-sonnet-4-6', 'claude-haiku-4-5'] as const
  export type AllowedModel = typeof ALLOWED_MODELS[number]

  interface Prefs { model: AllowedModel; vcScrapeInterval: string }

  async function load(): Promise<Prefs> {
    const p = prefsPath()
    if (!existsSync(p)) return { model: DEFAULT_MODEL, vcScrapeInterval: DEFAULT_VC_INTERVAL }
    try {
      const raw = await fs.readFile(p, 'utf-8')
      const json = JSON.parse(raw)
      const model = ALLOWED_MODELS.includes(json.model) ? json.model : DEFAULT_MODEL
      const vcScrapeInterval =
        typeof json.vcScrapeInterval === 'string' && cron.validate(json.vcScrapeInterval)
          ? json.vcScrapeInterval
          : DEFAULT_VC_INTERVAL
      return { model, vcScrapeInterval }
    } catch {
      return { model: DEFAULT_MODEL, vcScrapeInterval: DEFAULT_VC_INTERVAL }
    }
  }

  export const preferences = {
    async getModel(): Promise<AllowedModel> { return (await load()).model },
    async setModel(model: string): Promise<void> {
      if (!ALLOWED_MODELS.includes(model as AllowedModel)) throw new Error(`Unsupported model: ${model}`)
      const current = await load()
      await writeFileAtomic(prefsPath(), JSON.stringify({ ...current, model }, null, 2))
    },
    async getVcScrapeInterval(): Promise<string> { return (await load()).vcScrapeInterval },
    async setVcScrapeInterval(expr: string): Promise<void> {
      if (!cron.validate(expr)) throw new Error('Invalid cron expression')
      const current = await load()
      await writeFileAtomic(prefsPath(), JSON.stringify({ ...current, vcScrapeInterval: expr }, null, 2))
    },
    DEFAULT_MODEL,
    DEFAULT_VC_INTERVAL,
    ALLOWED_MODELS,
  }
  ```

  ```typescript
  // HEAD probe for Add Firm — verifies a portfolio URL is reachable before saving.
  // SSRF guard: only http(s), no private-IP literals in host, 5s timeout.

  const PRIVATE_HOST_RE = /^(localhost|127\.|0\.0\.0\.0|10\.|192\.168\.|169\.254\.|::1|fc[0-9a-f]{2}:|fd[0-9a-f]{2}:)/i

  export interface ProbeResult {
    ok: boolean
    status?: number
    error?: string
  }

  export async function probeUrl(rawUrl: string): Promise<ProbeResult> {
    let u: URL
    try { u = new URL(rawUrl) } catch { return { ok: false, error: 'invalid url' } }
    if (u.protocol !== 'http:' && u.protocol !== 'https:') {
      return { ok: false, error: 'scheme must be http or https' }
    }
    if (PRIVATE_HOST_RE.test(u.hostname)) {
      return { ok: false, error: 'private/loopback host not allowed' }
    }
    try {
      const res = await fetch(u.toString(), {
        method: 'HEAD',
        redirect: 'follow',
        signal: AbortSignal.timeout(5000),
        headers: { 'User-Agent': 'JobEngineBot/1.0' },
      })
      return { ok: res.ok, status: res.status }
    } catch (err: any) {
      return { ok: false, error: err?.name === 'TimeoutError' ? 'timeout' : (err?.message ?? 'probe failed') }
    }
  }
  ```

  ```typescript
        if (parts[firmIdx] === firm && parts[companyIdx] === company) {

          lines[i] = parts.join('\t')

      }
      return lines.join('\n')
  ```

  ```typescript
  // Electron-side cron scheduler for scrape-vcs.mjs.
  // Uses node-cron 4.2.1; a single in-flight lock (isScrapeActive) prevents overlap
  // between scheduled runs and manual triggers.

  import cron, { type ScheduledTask } from 'node-cron'
  import type { BrowserWindow } from 'electron'
  import { startOp } from './process-runner'
  import { preferences } from './preferences'

  let currentTask: ScheduledTask | null = null
  let scrapeActive = false

  export function isScrapeActive(): boolean {
    return scrapeActive
  }

  // Public entry used by ipc-handlers runVcScrape and by the cron callback.
  // Returns runId (spawn succeeded) or null (busy — caller should inform user).
  export function triggerScrape(projectRoot: string, win: BrowserWindow): string | null {
    if (scrapeActive) return null
    scrapeActive = true
    const runId = startOp({
      kind: 'scrape',
      command: 'node',
      args: ['scrape-vcs.mjs'],
      cwd: projectRoot,
      win,
    })
    // Release the lock when process-runner emits op:done. We subscribe once, per run.
    // process-runner sends on 'op:done' channel; we listen on the main-side `win.webContents`
    // by wrapping the lifecycle: rely on a small timer fallback + ipc side-channel.
    // Simpler: clear the flag when the activeOps count for this runId drops. Since startOp
    // does not return a promise, we attach a one-shot listener on the webContents IPC send
    // by watching for the renderer-bound 'op:done' event via a main-side observer.
    //
    // Implementation: use a short poll tick on activeOpsCount + runId match via a closure.
    const release = (): void => { scrapeActive = false }
    // process-runner already logs to the window; we add a side-channel via the 'op:done'
    // ipcMain listener below (see initScheduler). If no side-channel is wired, a 60-minute
    // safety timer releases the lock to avoid permanent stickiness.
    const safetyTimer = setTimeout(release, 60 * 60 * 1000)
    // Store the release + timer on a module map keyed by runId so initScheduler's
    // op:done listener can call it.
    pendingReleases.set(runId, () => { clearTimeout(safetyTimer); release() })
    return runId
  }

  // Map of runId -> release function. Wired up by initScheduler's ipcMain listener.
  const pendingReleases = new Map<string, () => void>()

  export async function initScheduler(projectRoot: string, win: BrowserWindow): Promise<void> {
    // Wire a side-channel: process-runner sends 'op:done' to the window; we intercept
    // the same event on the main side via a webContents 'did-finish-load'-free approach:
    // re-emit via ipcMain is not possible, so we attach an IPC-main listener on a new
    // dedicated channel 'scrape:done'. Easier path: modify startOp? No — keep it generic.
    //
    // Pragmatic solution: listen for the webContents 'ipc-message' event on the outbound
    // 'op:done' send. Electron's webContents does not expose outbound listeners, so we
    // instead attach a dedicated cleanup hook by wrapping triggerScrape's caller: the cron
    // callback awaits via a polling check on pendingReleases. Simplest: clear the flag
    // when the renderer echoes the op:done event back via a new 'scrape:ack' channel.
    //
    // For reliability, we poll activeOpsCount every 2s and release when it hits zero AND
    // we had a pending release. This is robust because pdf/scan/batch ops are rare and the
    // lock is only for scrape.
    const { activeOpsCount } = await import('./process-runner')
    setInterval(() => {
      if (pendingReleases.size === 0) return
      if (activeOpsCount() === 0) {
        for (const release of pendingReleases.values()) release()
        pendingReleases.clear()
      }
    }, 2000).unref()

    const expr = await preferences.getVcScrapeInterval()
    currentTask = cron.schedule(expr, () => { triggerScrape(projectRoot, win) }, { scheduled: true })
  }

  export async function reconfigureScheduler(projectRoot: string, win: BrowserWindow): Promise<void> {
    if (currentTask) { currentTask.stop() }
    const expr = await preferences.getVcScrapeInterval()
    currentTask = cron.schedule(expr, () => { triggerScrape(projectRoot, win) }, { scheduled: true })
  }

  export function stopScheduler(): void {
    if (currentTask) { currentTask.stop(); currentTask = null }
    pendingReleases.clear()
    scrapeActive = false
  }
  ```

  Note on the scheduler lock strategy: since we cannot intercept outbound webContents.send events from the main side, we use `activeOpsCount() === 0` polling as the completion signal. This is safe because (a) the scrape op is long-running and rare, (b) activeOpsCount is a module-local Map in process-runner.ts, and (c) the 60-minute safety timer is a hard backstop.
