  ```typescript
  import { useEffect, useRef, useState } from 'react'

  export interface StatusOption {
    label: string
    value: string
    category: 'Active' | 'Completed' | 'Outcome'
    color?: string
  }

  export interface StatusSelectProps {
    currentStatus: string
    options: StatusOption[]
    onSave: (newStatus: string) => Promise<void>
    onCancel: () => void
    autoFocus?: boolean
  }

  function groupByCategory(options: StatusOption[]): Record<string, StatusOption[]> {
    const groups: Record<string, StatusOption[]> = { Active: [], Completed: [], Outcome: [] }
    for (const opt of options) {
      if (!groups[opt.category]) groups[opt.category] = []
      groups[opt.category].push(opt)
    }
    return groups
  }

  export function StatusSelect({
    currentStatus,
    options,
    onSave,
    onCancel,
    autoFocus = true,
  }: StatusSelectProps) {
    const ref = useRef<HTMLSelectElement>(null)
    const [saving, setSaving] = useState(false)
    const [value, setValue] = useState(currentStatus)

    useEffect(() => {
      if (autoFocus && ref.current) {
        ref.current.focus()
      }
    }, [autoFocus])

    useEffect(() => {
      const handler = (e: KeyboardEvent) => {
        if (e.key === 'Escape' && !saving) {
          e.preventDefault()
          onCancel()
        }
      }
      window.addEventListener('keydown', handler)
      return () => window.removeEventListener('keydown', handler)
    }, [onCancel, saving])

    const handleChange = async (e: React.ChangeEvent<HTMLSelectElement>) => {
      const next = e.target.value
      setValue(next)
      if (next === currentStatus) {
        onCancel()
        return
      }
      setSaving(true)
      try {
        await onSave(next)
      } finally {
        setSaving(false)
      }
    }

    const handleBlur = () => {
      if (saving) return
      // Defer: clicking within the same React update (e.g., own option) shouldn't cancel.
      // Microtask delay lets onChange take precedence.
      queueMicrotask(() => {
        if (!saving) onCancel()
      })
    }

    const groups = groupByCategory(options)

    return (
      <span className="status-select" data-testid="status-select" data-saving={saving}>
        <select
          ref={ref}
          className="status-select__input"
          value={value}
          onChange={handleChange}
          onBlur={handleBlur}
          disabled={saving}
          aria-label="Change status"
        >
          {(Object.keys(groups) as Array<'Active' | 'Completed' | 'Outcome'>).map((cat) =>
            groups[cat].length > 0 ? (
              <optgroup key={cat} label={cat}>
                {groups[cat].map((opt) => (
                  <option key={opt.value} value={opt.value}>{opt.label}</option>
                ))}
              </optgroup>
            ) : null,
          )}
        </select>
        {saving && <span className="status-select__spinner" aria-hidden="true">⏳</span>}
      </span>
    )
  }
  ```

  ```typescript
  import { useEffect, useState } from 'react'

  interface Props {
    message: string
    /** Timestamp (ms) used as key to retrigger animation; null when hidden. */
    at: number | null
    durationMs?: number
    onDismiss: () => void
  }

  export function StatusUpdateToast({ message, at, durationMs = 2500, onDismiss }: Props) {
    const [visible, setVisible] = useState(false)

    useEffect(() => {
      if (at === null) { setVisible(false); return }
      setVisible(true)
      const timer = setTimeout(() => {
        setVisible(false)
        onDismiss()
      }, durationMs)
      return () => clearTimeout(timer)
    }, [at, durationMs, onDismiss])

    if (!visible) return null
    return (
      <div
        className="status-toast"
        role="status"
        data-testid="status-toast"
        onClick={() => { setVisible(false); onDismiss() }}
      >
        {message}
      </div>
    )
  }
  ```

  ```typescript
  import { useEffect, useState } from 'react'
  import { TrackerRow } from './TrackerRow'
  import { StatusUpdateToast } from './StatusUpdateToast'
  import type { StatusOption } from './StatusSelect'
  import type { TrackerRow as TrackerRowType } from '../../preload/types'
  import { EmptyState } from './EmptyState'
  import { ErrorState } from './ErrorState'

  interface Props {
    onOpenReport?: (reportPath: string) => void
  }

  export function TrackerPanel({ onOpenReport }: Props) {
    const [rows, setRows] = useState<TrackerRowType[]>([])
    const [statusOptions, setStatusOptions] = useState<StatusOption[]>([])
    const [loading, setLoading] = useState(true)
    const [error, setError] = useState<string | null>(null)
    const [activeEditRow, setActiveEditRow] = useState<number | null>(null)
    const [toast, setToast] = useState<{ message: string; at: number } | null>(null)

    const reload = async () => {
      try {
        const [tracker, statuses] = await Promise.all([
          window.api.readTracker(),
          window.api.readStatuses(),
        ])
        setRows(tracker)
        // Map StatusEntry[] from preload to StatusOption[]; preserve Phase 1 field names.
        setStatusOptions(
          statuses.map((s) => ({
            label: s.label,
            value: s.value,
            category: (s.category ?? 'Active') as StatusOption['category'],
            color: s.color,
          })),
        )
        setError(null)
      } catch (err: any) {
        setError(err?.message ?? 'Failed to load tracker')
      } finally {
        setLoading(false)
      }
    }

    useEffect(() => {
      void reload()
      const unsub = window.api.onFilesChanged(() => {
        void reload()
      })
      return () => unsub()
    }, [])

    const handleStartEdit = (num: number) => {
      setActiveEditRow(num)
    }

    const handleCancelEdit = () => {
      setActiveEditRow(null)
    }

    const handleSave = async (num: number, newStatus: string) => {
      const result = await window.api.updateStatus(num, newStatus)
      if (result.success) {
        setActiveEditRow(null)
        setToast({ message: `Status updated to ${newStatus}`, at: Date.now() })
        // rows will re-populate via onFilesChanged → reload()
      } else {
        setToast({
          message: result.message ?? result.error ?? 'Failed to update status',
          at: Date.now(),
        })
        // Keep activeEditRow so the user can retry
      }
    }

    if (loading) return <EmptyState message="Loading tracker..." />
    if (error) return <ErrorState message={error} onRetry={reload} />
    if (rows.length === 0) return <EmptyState message="No applications yet." />

    return (
      <section className="tracker-panel" data-testid="tracker-panel">
        <div className="tracker-panel__body">
          <table className="tracker-table">
            <thead>
              <tr>
                <th>#</th><th>Date</th><th>Company</th><th>Role</th>
                <th>Score</th><th>Status</th><th>PDF</th><th>Report</th><th>Notes</th>
              </tr>
            </thead>
            <tbody>
              {rows.map((row) => (
                <TrackerRow
                  key={row.num}
                  row={row}
                  statusOptions={statusOptions}
                  isEditing={activeEditRow === row.num}
                  onStartEdit={() => handleStartEdit(row.num)}
                  onSave={(newStatus) => handleSave(row.num, newStatus)}
                  onCancelEdit={handleCancelEdit}
                  onOpenReport={onOpenReport}
                />
              ))}
            </tbody>
          </table>
        </div>
        <StatusUpdateToast
          message={toast?.message ?? ''}
          at={toast?.at ?? null}
          onDismiss={() => setToast(null)}
        />
      </section>
    )
  }
  ```
